📋 WHAT IS CIXIO?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CIXIO is a comprehensive full-stack application with:
  ✓ Backend REST API (Express.js + Node.js)
  ✓ MongoDB Database (4 models)
  ✓ 22 API endpoints
  ✓ JWT Authentication & Authorization
  ✓ Professional Email Service (7 templates)
  ✓ User Management System
  ✓ Subscription Management
  ✓ Contact Form System
  ✓ Newsletter Management (Double opt-in)
  ✓ Enterprise-grade Security Features

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
⚡ QUICK START (5 MINUTES)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. INSTALL DEPENDENCIES
   $ npm install

2. CONFIGURE ENVIRONMENT
   $ cp .env.example .env
   $ nano .env    # Edit configuration (see below)

3. START MONGODB (if running locally)
   $ mongod --dbpath /path/to/data

4. START THE SERVER
   $ npm start                # Production
   $ npm run dev             # Development (with auto-reload)

5. TEST THE APPLICATION
   Open: http://localhost:3000/api/health

🔐 ESSENTIAL CONFIGURATION (.env file)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

MINIMUM REQUIRED SETTINGS:

# Database (choose one):
MONGODB_URI=mongodb://localhost:27017/cixio                          # Local
MONGODB_URI=mongodb+srv://<cluster-host>/cixio        # Atlas

# Security (IMPORTANT - Generate secure key!)
JWT_SECRET=your-secure-random-key-here    # Run: node -e "console.log(require('crypto').randomBytes(64).toString('hex'))"
JWT_EXPIRE=7d

# Email Service (for www.cixio.com domain)
EMAIL_HOST=mail.cixio.com
EMAIL_PORT=587
EMAIL_USER=noreply@cixio.com
EMAIL_PASS=your-email-password
EMAIL_FROM=CIXIO <noreply@cixio.com>

# Application URLs
FRONTEND_URL=https://www.cixio.com
API_URL=https://www.cixio.com/api
